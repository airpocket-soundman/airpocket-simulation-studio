cmake_minimum_required(VERSION 3.20)
project(AirpocketNativeProof LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(MFEM_USE_MPI OFF CACHE BOOL "" FORCE)
set(MFEM_USE_METIS OFF CACHE BOOL "" FORCE)
set(MFEM_USE_LAPACK OFF CACHE BOOL "" FORCE)
set(MFEM_USE_OPENMP OFF CACHE BOOL "" FORCE)
set(MFEM_USE_EXCEPTIONS ON CACHE BOOL "" FORCE)
set(MFEM_PRECISION double CACHE STRING "" FORCE)
set(MFEM_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(MFEM_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(MFEM_ENABLE_MINIAPPS OFF CACHE BOOL "" FORCE)
if(NOT MFEM_SOURCE_DIR)
  message(FATAL_ERROR "Set MFEM_SOURCE_DIR to the existing pinned MFEM source cache")
endif()
add_subdirectory(${MFEM_SOURCE_DIR} mfem-core)
add_executable(airpocket-native-proof main.cpp ../mfem-adapter/adapter.cpp)
target_include_directories(airpocket-native-proof PRIVATE compat)
target_link_libraries(airpocket-native-proof PRIVATE mfem)
