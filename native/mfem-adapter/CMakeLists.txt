cmake_minimum_required(VERSION 3.20)
project(AirpocketMfem LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(MFEM_USE_MPI OFF CACHE BOOL "" FORCE)
set(MFEM_USE_METIS OFF CACHE BOOL "" FORCE)
set(MFEM_USE_LAPACK OFF CACHE BOOL "" FORCE)
set(MFEM_USE_OPENMP OFF CACHE BOOL "" FORCE)
set(MFEM_USE_EXCEPTIONS ON CACHE BOOL "" FORCE)
set(MFEM_PRECISION double CACHE STRING "" FORCE)
set(MFEM_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(MFEM_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(MFEM_ENABLE_MINIAPPS OFF CACHE BOOL "" FORCE)
add_subdirectory(${MFEM_SOURCE_DIR} mfem-core)
add_executable(airpocket-mfem adapter.cpp)
target_link_libraries(airpocket-mfem PRIVATE mfem)
target_link_options(airpocket-mfem PRIVATE --no-entry -sMODULARIZE=1 -sEXPORT_ES6=1 -sEXPORT_NAME=createMfem -sENVIRONMENT=web,worker,node -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=536870912 -sINITIAL_MEMORY=33554432 -sSTACK_SIZE=5242880 -sEXPORTED_FUNCTIONS=_solve_elastic,_last_error,_mfem_version,_adapter_abi,_malloc,_free -sEXPORTED_RUNTIME_METHODS=UTF8ToString,HEAPU8,HEAPF64 -fexceptions -msimd128)
set_target_properties(airpocket-mfem PROPERTIES OUTPUT_NAME mfem SUFFIX .mjs RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../public/mfem")
