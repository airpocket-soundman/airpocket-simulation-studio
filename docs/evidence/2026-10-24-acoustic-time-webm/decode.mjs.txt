import fs from 'node:fs';import path from 'node:path';import {execFileSync} from 'node:child_process';import {createHash} from 'node:crypto';import assert from 'node:assert/strict';const root='.qa/acoustic-webm191/out',file=path.join(root,'acoustic.webm'),sha=b=>createHash('sha256').update(b).digest('hex'),run=(exe,args)=>execFileSync(exe,args,{maxBuffer:8*1024*1024});const probe=JSON.parse(run('ffprobe',['-v','error','-select_streams','v:0','-show_streams','-show_format','-show_frames','-of','json',file]));fs.writeFileSync(path.join(root,'ffprobe.json'),JSON.stringify(probe,null,2));const frames=probe.frames.filter(f=>f.media_type==='video'),stream=probe.streams[0],status=JSON.parse(fs.readFileSync(path.join(root,'acoustic-raw-completed.json'))),m=status.status.artifact.manifest;assert.ok(frames.length>=2&&frames.length<=2048);assert.equal(stream.width,m.resolution.width);assert.equal(stream.height,m.resolution.height);assert.ok(['vp8','vp9'].includes(stream.codec_name));const indexes=[0,Math.floor((frames.length-1)/2),frames.length-1];for(let i=0;i<3;i++)run('ffmpeg',['-v','error','-i',file,'-vf',`select=eq(n\\,${indexes[i]})`,'-frames:v','1','-fps_mode','passthrough','-y',path.join(root,'decoded-'+i+'.png')]);const hashes=run('ffmpeg',['-v','error','-i',file,'-map','0:v:0','-fps_mode','passthrough','-f','framehash','-hash','sha256','-']);fs.writeFileSync(path.join(root,'frames.sha256'),hashes);const lines=hashes.toString().split('\n').filter(s=>s&&!s.startsWith('#'));assert.equal(lines.length,frames.length);const unique=new Set(lines.map(l=>l.split(',').at(-1).trim()));assert.ok(unique.size>=2);const pts=frames.map(f=>Number(f.best_effort_timestamp_time));assert.ok(pts.every((p,i)=>Number.isFinite(p)&&(!i||p>=pts[i-1])));const report={decoder:run('ffmpeg',['-version']).toString().split('\n')[0],probe:run('ffprobe',['-version']).toString().split('\n')[0],bytes:fs.statSync(file).size,sha256:sha(fs.readFileSync(file)),codec:stream.codec_name,width:stream.width,height:stream.height,frameCount:frames.length,distinctFrameHashes:unique.size,presentationTimesSeconds:pts,formatDurationSeconds:probe.format.duration??null,selectedFrames:indexes.map((index,i)=>({decodedIndex:index,ptsSeconds:pts[index],file:'decoded-'+i+'.png'})),checksPassed:true,limitations:['Variable real-time recording; no fixed frame count or exact24fps claim.','Presentation PTS is not physical time; no per-encoded-frame physical ledger contract.']};fs.writeFileSync(path.join(root,'independent-decode.json'),JSON.stringify(report,null,2));console.log(JSON.stringify(report));
